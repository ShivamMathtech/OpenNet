cmake_minimum_required(VERSION 3.16)
project(OpenNet VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
add_executable(opennet-node apps/node.cpp)
target_include_directories(opennet-node PRIVATE core)
target_link_libraries(opennet-node PRIVATE OpenSSL::Crypto)
target_compile_options(opennet-node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(packet-tests tests/packet_test.cpp)
target_include_directories(packet-tests PRIVATE core)
target_link_libraries(packet-tests PRIVATE OpenSSL::Crypto)
enable_testing()
add_test(NAME packet-tests COMMAND packet-tests)
